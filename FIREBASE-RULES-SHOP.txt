অনলাইন দোকানের জন্য নতুন Firebase Rules (সংস্করণ ৫.৮)
==========================================================
এটা আপনার আগের Rules-এর সাথে যোগ করতে হবে (বদলে নয়)। Firestore Database → Rules ট্যাবে আগের নিয়মের
`match /databases/{database}/documents { ... }` ব্লকের ভেতরে, শেষ `}`-এর ঠিক আগে নিচের ৩টি match ব্লক বসিয়ে Publish করুন।
(আগের নিয়মে hasRole() ফাংশন আছে — সাময়িক ও চূড়ান্ত দুই ধাপেই। না থাকলে নিচের ফাংশনটিও যোগ করুন।)

    // ---- অনলাইন দোকান (নতুন) ----

    // কাস্টমার-দেখা পণ্য তালিকা ও ছবি: সবাই পড়তে পারে, লিখতে পারে শুধু লগইন করা স্টাফ
    match /averamart_public/{d} {
      allow read: if true;
      allow write: if hasRole();
    }
    match /averamart_public_img/{d} {
      allow read: if true;
      allow write: if hasRole();
    }

    // ওয়েব অর্ডার: যে কেউ শুধু নতুন অর্ডার "তৈরি" করতে পারে (নির্দিষ্ট ছকে); পড়া/বদলানো শুধু স্টাফ; মোছা যায় না
    match /averamart_weborders/{id} {
      allow create: if request.resource.data.keys().hasOnly(['customerName','phone','address','note','area','items','subtotal','delivery','deliveryTbd','total','status','source','createdAt'])
        && request.resource.data.keys().hasAll(['customerName','phone','address','items','total','status','createdAt'])
        && request.resource.data.customerName is string && request.resource.data.customerName.size() > 0 && request.resource.data.customerName.size() <= 80
        && request.resource.data.phone is string && request.resource.data.phone.matches('^01[0-9]{9}$')
        && request.resource.data.address is string && request.resource.data.address.size() >= 8 && request.resource.data.address.size() <= 300
        && (!('note' in request.resource.data) || (request.resource.data.note is string && request.resource.data.note.size() <= 300))
        && request.resource.data.items is list && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 40
        && request.resource.data.total is number && request.resource.data.total >= 0 && request.resource.data.total < 1000000
        && request.resource.data.status == 'new'
        && request.resource.data.createdAt == request.time;
      allow read: if hasRole();
      allow update: if hasRole() && resource.data.status == 'new'
        && request.resource.data.status in ['imported','cancelled']
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','orderId','handledBy','handledAt']);
      allow delete: if false;
    }

জরুরি জেনে রাখুন
  • এই ব্লক ছাড়া দোকান পেইজে পণ্য দেখাবে না ও অর্ডার যাবে না; অ্যাডমিন অ্যাপে "সিঙ্ক হয়নি" লেখা আসবে।
  • দোকান পেইজ লগইন ছাড়া চলে, তাই কেউ অর্ডার-ছক মেনে ভুয়া অর্ডার পাঠাতে পারে। ফোনে কনফার্ম করে "গ্রহণ করুন" চাপার নিয়মই এর প্রধান সুরক্ষা।
    ভুয়া অর্ডার বেশি এলে Firebase Console → App Check চালু করুন (reCAPTCHA)।
  • কাস্টমারের নাম/ফোন/ঠিকানা averamart_weborders-এ থাকে; পড়তে পারে শুধু আপনার স্টাফ।
  • পণ্যের ক্রয়মূল্য ও লাভ কখনো averamart_public-এ যায় না।
